#include <tunables/global>

# BKON LightRAG add-on AppArmor profile.
#
# Runtime: python:3.11-slim base (no s6-overlay; init: false). run.sh reads the
# add-on options into namespaced env vars and execs uvicorn, which serves the
# LightRAG API (POST /query, /documents/text, GET /health, and the ingress
# status page at /). The service embeds locally (fastembed/ONNX, CPU) and makes
# outbound TCP calls to the selected generation provider (Ollama / Anthropic /
# OpenAI-compatible). Its data — the RAG graph and vector store — lives under
# /share and /data.
#
# Strategy mirrors the sibling add-ons: allow the app's own file I/O (broad
# `file` rule) and let AppArmor's default-deny confine everything else. Every
# capability, mount, module load, ptrace and raw socket NOT granted below is
# denied. This service is narrower than Edibl — no privilege drop, no Supervisor
# API — so it asks for less: no setuid/setgid/gosu, no hassio access.
profile bkon_lightrag flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # The app reads/writes its own files: code in /app, the ONNX embedding model
  # cache, the RAG storage under /share and /data, /tmp, /proc, and the Python
  # runtime / site-packages.
  file,

  # Signals: uvicorn's own process management and clean shutdown forwarding.
  signal,

  # Execute the entrypoint, interpreters and helpers within THIS profile
  # (ix = inherit), so uvicorn/python3 stay confined by the same rules.
  /run.sh ix,
  /app/** ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/local/bin/** ix,

  # Networking: serve the API over TCP, reach the generation provider (TCP
  # egress), resolve DNS (UDP + netlink), and local sockets. No raw packet
  # sockets — the service never needs them.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network netlink raw,
  network unix stream,
  network unix dgram,
}
